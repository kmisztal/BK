\section{IWRS / IVRS}

% SLIDE ID: sec05-goal
\BlockGoalFrame{\SlideID{sec05-goal}}{
  \item zrozumieć rolę systemów randomizacji w badaniu klinicznym,
  \item zobaczyć, jak system zarządza przypisaniem leku badanego,
  \item omówić ryzyka operacyjne i skutki błędów użytkownika.
}

% SLIDE ID: sec05-ivrs
\begin{frame}{Czym jest IVRS}
  \SlideID{sec05-ivrs}
  \begin{itemize}
    \item IVRS to telefoniczny system interaktywny używany historycznie i nadal w części badań,
    \item użytkownik wykonuje operacje przez połączenie głosowe i wybór opcji,
    \item system wspiera randomizację i logistykę leku bez konieczności pracy w interfejsie webowym.
  \end{itemize}
\end{frame}

% SLIDE ID: sec05-iwrs
\begin{frame}{Czym jest IWRS}
  \SlideID{sec05-iwrs}
  \begin{itemize}
    \item IWRS to internetowy system randomizacji i zarządzania lekiem badanym,
    \item zwykle przejmuje funkcje IVRS w nowoczesnych badaniach,
    \item łączy logikę randomizacji, zapasy leku i zapis operacji w jednym miejscu.
  \end{itemize}
\end{frame}

% SLIDE ID: sec05-history-overview
\begin{frame}{Historia rozwoju IVRS / IWRS / RTSM}
  \SlideID{sec05-history-overview}
  \begin{center}
    \Large
    Ręczne zarządzanie $\rightarrow$ telefoniczne \texttt{IVRS} $\rightarrow$ webowe \texttt{IWRS} $\rightarrow$ współczesne \texttt{RTSM / IRT}
  \end{center}

  \vspace{0.45cm}

  \begin{itemize}
    \item Historia tych systemów pokazuje, że randomizacja i logistyka leku nie mogły pozostać procesami ręcznymi przy rosnącej złożoności badań.
    \item Z czasem ciężar systemu przesunął się z samego przypisania leczenia na szersze zarządzanie zapasem, resupply i kontrolę nad całym przepływem leku badanego.
    \item Dlatego współczesne \texttt{IWRS} coraz częściej są opisywane szerzej jako \texttt{RTSM} albo \texttt{IRT}.
  \end{itemize}
\end{frame}

% SLIDE ID: sec05-history-pre-digital
\begin{frame}{1. Era przedcyfrowa: przed latami 90.}
  \SlideID{sec05-history-pre-digital}
  \begin{itemize}
    \item Przed automatyzacją personel ośrodka i zespoły operacyjne zarządzały randomizacją oraz lekiem badanym ręcznie.
    \item Zestawy leku były wysyłane do ośrodków z góry, często w pełnych blokach odpowiadających planowanemu leczeniu pacjentów.
    \item Jeśli ośrodek rekrutował słabo albo pacjenci odpadali z badania, rosło marnotrawstwo zapasu i ryzyko nieefektywnej dystrybucji.
    \item Ten model był mało elastyczny, słabo skalowalny i utrudniał utrzymanie bieżącej kontroli nad dostępnością leku.
  \end{itemize}
\end{frame}

% SLIDE ID: sec05-history-ivrs
\begin{frame}{2. Wzrost znaczenia IVRS: lata 90.}
  \SlideID{sec05-history-ivrs}
  \begin{itemize}
    \item W latach 90. pojawiły się pierwsze systemy \texttt{IVRS}, oparte na połączeniach telefonicznych i wyborach z klawiatury numerycznej.
    \item Użytkownik dzwonił na numer systemu, wprowadzał dane pacjenta i otrzymywał przydział leczenia lub numer zestawu w czasie rzeczywistym.
    \item Najważniejszą korzyścią była automatyzacja randomizacji bez ujawniania przydziału leczenia osobom, które nie powinny go znać.
    \item Z czasem \texttt{IVRS} zaczęło wspierać nie tylko randomizację, ale także dispensing i podstawowy resupply na poziomie ośrodka.
  \end{itemize}
\end{frame}

% SLIDE ID: sec05-history-iwrs
\begin{frame}{3. Rewolucja webowa: od IWRS do RTSM}
  \SlideID{sec05-history-iwrs}
  \begin{itemize}
    \item Wraz z upowszechnieniem internetu pojawiły się webowe interfejsy \texttt{IWRS}, które początkowo bywały dodatkiem lub backupem dla rozwiązań telefonicznych.
    \item Z czasem to właśnie interfejs webowy stał się dominującym standardem pracy operacyjnej w badaniach klinicznych.
    \item Kluczowa zmiana dotyczyła logistyki: system przestał tylko losować pacjenta i zaczął aktywnie sterować zapasem, resupply i ruchem leku między depot a ośrodkiem.
    \item Dlatego coraz częściej mówi się o \texttt{RTSM} (\texttt{Randomization and Trial Supply Management}) albo szerzej o \texttt{IRT} (\texttt{Interactive Response Technology}) jako parasolowym terminie dla tych technologii.
  \end{itemize}
\end{frame}

% SLIDE ID: sec05-history-modern
\begin{frame}{4. Współczesne systemy IWRS / RTSM / IRT}
  \SlideID{sec05-history-modern}
  \begin{itemize}
    \item Dzisiejsze systemy wspierają nie tylko randomizację i przydział leku, ale również forecasting, zarządzanie zapasem, expiry management i ścieżki wyjątków.
    \item Integrują się z \texttt{EDC}, depotami, systemami dostaw i innymi elementami ekosystemu \texttt{eClinical}.
    \item W bardziej złożonych badaniach mogą obsługiwać adaptacyjne schematy randomizacji, kontrolę temperatury, wymiany zestawów i złożoną logistykę biologiczną.
    \item W praktyce są dziś technologicznymi hubami łączącymi decyzję randomizacyjną z realnym przepływem leku badanego.
  \end{itemize}
\end{frame}

% SLIDE ID: sec05-history-what-changed
\begin{frame}{Dlaczego ta ewolucja ma znaczenie}
  \SlideID{sec05-history-what-changed}
  \begin{itemize}
    \item Zmieniła precyzję i bezstronność randomizacji, bo logika systemu zastąpiła proces ręczny.
    \item Zmieniła ekonomię badania, bo lek nie musi być wysyłany „na zapas” w takim stopniu jak dawniej.
    \item Zmieniła bezpieczeństwo operacyjne, bo łatwiej śledzić wydanie, zwrot, zniszczenie i odchylenia od planu.
    \item Zmieniła też zakres odpowiedzialności: nowoczesny \texttt{IWRS} to nie tylko narzędzie do przydziału ramienia, ale centrum sterowania supply chain badania.
  \end{itemize}
\end{frame}

% SLIDE ID: sec05-randomization
\begin{frame}{Randomizacja, stratyfikacja, zaślepienie}
  \SlideID{sec05-randomization}
  \begin{itemize}
    \item randomizacja przypisuje pacjenta do ramienia badania według ustalonych reguł,
    \item stratyfikacja pomaga utrzymać równowagę między grupami,
    \item zaślepienie chroni badanie przed ujawnieniem przydziału leczenia.
  \end{itemize}
\end{frame}

% SLIDE ID: sec05-randomization-types-overview
\begin{frame}{Najczęstsze rodzaje randomizacji}
  \SlideID{sec05-randomization-types-overview}
  \begin{itemize}
    \item randomizacja prosta,
    \item randomizacja blokowa,
    \item randomizacja stratyfikowana,
    \item randomizacja adaptacyjna,
    \item minimalizacja.
  \end{itemize}

  \vspace{0.25cm}

  \begin{ExampleBox}
    Wybór metody zależy od projektu badania, liczby ramion, wielkości próby i potrzeby utrzymania równowagi między grupami.
  \end{ExampleBox}
\end{frame}

% SLIDE ID: sec05-simple-randomization
\begin{frame}{Randomizacja prosta}
  \SlideID{sec05-simple-randomization}
  \begin{itemize}
    \item każdy pacjent ma niezależną szansę przypisania do ramienia badania,
    \item klasyczny przykład: proporcja 1:1 dla placebo vs lek badany,
    \item największą zaletą jest prostota implementacji i zrozumienia,
    \item ryzyko polega na tym, że przy małej próbie grupy mogą okazać się nierówne liczebnie.
  \end{itemize}
\end{frame}

% SLIDE ID: sec05-block-randomization
\begin{frame}{Randomizacja blokowa}
  \SlideID{sec05-block-randomization}
  \begin{itemize}
    \item pacjenci są randomizowani w blokach, np. po 4 lub 6 osób,
    \item w każdym bloku zachowana jest proporcja między ramionami,
    \item przykłady bloków 1:1: \texttt{AABB}, \texttt{ABAB}, \texttt{BAAB},
    \item zaleta: łatwiej utrzymać równowagę liczebności grup,
    \item ryzyko: przy małych i stałych blokach może rosnąć przewidywalność.
  \end{itemize}
\end{frame}

% SLIDE ID: sec05-stratified-randomization
\begin{frame}{Randomizacja stratyfikowana}
  \SlideID{sec05-stratified-randomization}
  \begin{itemize}
    \item osobne listy randomizacyjne tworzy się dla ważnych cech pacjenta lub ośrodka,
    \item przykłady warstw: ośrodek, płeć, grupa wiekowa, stadium choroby,
    \item dodatkowo można uwzględnić wcześniejsze leczenie lub wynik biomarkera,
    \item celem jest ograniczenie sytuacji, w której jedna grupa ma więcej pacjentów o lepszym lub gorszym rokowaniu.
  \end{itemize}
\end{frame}

% SLIDE ID: sec05-stratification-example
\begin{frame}{Przykład stratyfikacji}
  \SlideID{sec05-stratification-example}
  \small
  \begin{beamercolorbox}[wd=\linewidth,sep=0.8em,rounded=true]{block body}
    \textbf{\textcolor{UAFMDarkBlue}{Badanie onkologiczne 1:1}}\par\medskip
    lek badany vs placebo
  \end{beamercolorbox}

  \vspace{0.25cm}

  \begin{itemize}
    \item straty: stadium choroby \texttt{II / III},
    \item wcześniejsze leczenie: \texttt{tak / nie},
    \item ośrodek: \texttt{A / B / C}.
  \end{itemize}

  \vspace{0.25cm}

  \begin{ExampleBox}
    \texttt{IWRS} najpierw ustala kombinację straty, np. \texttt{stadium III + leczenie tak + ośrodek B}, a dopiero potem losuje pacjenta z odpowiedniej listy randomizacyjnej.
  \end{ExampleBox}
\end{frame}

% SLIDE ID: sec05-adaptive-randomization
\begin{frame}{Randomizacja adaptacyjna}
  \SlideID{sec05-adaptive-randomization}
  \begin{itemize}
    \item prawdopodobieństwo przypisania do ramienia może zmieniać się w trakcie badania,
    \item zmiana może zależeć od liczby pacjentów, dostępnych danych lub założeń statystycznych,
    \item taki model wymaga ścisłej kontroli, walidacji i zgodności z protokołem,
    \item nie jest to zwykły „ręczny wybór lepszego ramienia” przez zespół badania.
  \end{itemize}
\end{frame}

% SLIDE ID: sec05-minimization
\begin{frame}{Minimalizacja}
  \SlideID{sec05-minimization}
  \begin{itemize}
    \item jest to metoda dynamicznego przypisywania pacjenta tak, aby zmniejszać nierównowagę ważnych cech,
    \item przykładowe cechy: ośrodek, stadium choroby, płeć, biomarker,
    \item często zawiera element losowy, aby nie była w pełni przewidywalna,
    \item bywa szczególnie przydatna w mniejszych lub bardziej złożonych badaniach.
  \end{itemize}
\end{frame}

% SLIDE ID: sec05-iwrs-randomization-logic
\begin{frame}{Co faktycznie robi IWRS podczas randomizacji}
  \SlideID{sec05-iwrs-randomization-logic}
  \begin{itemize}
    \item sprawdza uprawnienia użytkownika,
    \item sprawdza kwalifikację pacjenta,
    \item odczytuje dane stratyfikacyjne,
    \item wybiera odpowiednią listę lub algorytm,
    \item przypisuje ramię badania,
    \item przypisuje numer zestawu leku,
    \item zapisuje pełny audit trail.
  \end{itemize}
\end{frame}

% SLIDE ID: sec05-randomization-errors
\begin{frame}{Typowe błędy przy randomizacji}
  \SlideID{sec05-randomization-errors}
  \begin{itemize}
    \item błędnie wprowadzona strata,
    \item randomizacja pacjenta przed pełną kwalifikacją,
    \item randomizacja duplikatu pacjenta,
    \item użycie nieaktualnej listy randomizacyjnej,
    \item błędna konfiguracja ramion lub proporcji,
    \item brak zgodności między randomizacją a dostępnością leku.
  \end{itemize}
\end{frame}

% SLIDE ID: sec05-teaching-question-randomization
\begin{frame}{Pytanie do studentów}
  \SlideID{sec05-teaching-question-randomization}
  \begin{QuestionBox}
    Dlaczego w małym badaniu randomizacja prosta może być ryzykowna?
  \end{QuestionBox}

  \vspace{0.25cm}

  \begin{itemize}
    \item nierówna liczba pacjentów w ramionach,
    \item nierówny rozkład czynników rokowania,
    \item większe ryzyko problemów interpretacyjnych.
  \end{itemize}
\end{frame}

% SLIDE ID: sec05-sample-size-basics
\begin{frame}{Jak wyznacza się minimalną liczbę pacjentów}
  \SlideID{sec05-sample-size-basics}
  \begin{itemize}
    \item liczebność badania wynika z planu statystycznego, a nie z samego systemu \texttt{IWRS},
    \item punkt wyjścia to \textbf{primary endpoint}, oczekiwany efekt, poziom \texttt{alpha}, moc badania i zmienność danych lub częstość zdarzeń,
    \item najczęściej planuje się \texttt{alpha = 0{,}05} i moc \texttt{80\%} albo \texttt{90\%},
    \item na końcu dolicza się zapas na wycofania, brak danych i inne straty obserwacji.
  \end{itemize}

  \vspace{0.2cm}

  \begin{ExampleBox}
    \texttt{IWRS} nie liczy sam, ilu pacjentów trzeba zrandomizować. System wdraża wcześniej ustalony schemat: proporcje ramion, stratyfikację, bloki, minimalizację i listy randomizacyjne.
  \end{ExampleBox}
\end{frame}

% SLIDE ID: sec05-sample-size-stratification
\begin{frame}{Co zmienia stratyfikacja}
  \SlideID{sec05-sample-size-stratification}
  \begin{itemize}
    \item stratyfikacja zwykle \textbf{nie oznacza}, że mnożymy próbę przez liczbę warstw,
    \item oznacza jednak, że trzeba rozsądnie ograniczyć liczbę czynników stratyfikacyjnych,
    \item jeśli warstw jest zbyt dużo, w części kombinacji będzie bardzo mało pacjentów i równowaga stanie się niestabilna,
    \item dlatego w praktyce wybiera się tylko najważniejsze czynniki rokownicze, a przy bardziej złożonych układach rozważa się minimalizację.
  \end{itemize}

  \vspace{0.2cm}

  \begin{WarningBox}
    Z perspektywy \texttt{IWRS} zbyt rozbudowana stratyfikacja oznacza więcej list, więcej konfiguracji i większe ryzyko błędu operacyjnego.
  \end{WarningBox}
\end{frame}

% SLIDE ID: sec05-sample-size-formula
\begin{frame}{Skąd bierze się liczba pacjentów: prosty wzór}
  \SlideID{sec05-sample-size-formula}
  \small
  Dla badania 1:1 z punktem końcowym binarnym można użyć klasycznego przybliżenia:

  \[
  n =
  \frac{
    \left(
      z_{1-\alpha/2}\sqrt{2\bar{p}(1-\bar{p})}
      +
      z_{1-\beta}\sqrt{p_1(1-p_1)+p_2(1-p_2)}
    \right)^2
  }{
    (p_2-p_1)^2
  }
  \]

  \vspace{0.1cm}

  \begin{itemize}
    \item \(n\) to liczba pacjentów \textbf{na ramię},
    \item \(p_1\) i \(p_2\) to oczekiwane odsetki odpowiedzi,
    \item \(\bar{p}=(p_1+p_2)/2\),
    \item \(z_{1-\alpha/2}\) odpowiada poziomowi istotności, a \(z_{1-\beta}\) mocy badania.
  \end{itemize}
\end{frame}

% SLIDE ID: sec05-sample-size-case
\begin{frame}{Case study: proste liczenie liczebności badania}
  \SlideID{sec05-sample-size-case}
  \begin{beamercolorbox}[wd=\linewidth,sep=0.8em,rounded=true]{block body}
    \textbf{\textcolor{UAFMDarkBlue}{Założenie badania}}\par\medskip
    badanie 1:1, lek badany vs placebo, odpowiedź kliniczna po 12 tygodniach
  \end{beamercolorbox}

  \vspace{0.25cm}

  \begin{itemize}
    \item oczekiwana odpowiedź: \texttt{50\%} w kontroli vs \texttt{70\%} w leczeniu,
    \item \texttt{alpha = 0{,}05}, moc \texttt{80\%}, alokacja \texttt{1:1},
    \item podstawienie: \(p_1=0{,}50\), \(p_2=0{,}70\), \(\bar{p}=0{,}60\), \(z_{1-\alpha/2}=1{,}96\), \(z_{1-\beta}=0{,}84\),
    \item wynik: około \textbf{93 pacjentów na ramię}, czyli \textbf{186 łącznie},
    \item przy założeniu \texttt{10\%} utraty danych lub wycofań plan zwiększa się do około \textbf{104 na ramię}, czyli \textbf{208 łącznie}.
  \end{itemize}
\end{frame}

% SLIDE ID: sec05-sample-size-case-stratification
\begin{frame}{Case study: co zmienia stratyfikacja}
  \SlideID{sec05-sample-size-case-stratification}
  \begin{ExampleBox}
    Stratyfikacja po \texttt{ośrodku (A/B/C)} i \texttt{wcześniejszym leczeniu (tak/nie)} daje \texttt{6} warstw. Przy \texttt{208} pacjentach to nadal wykonalne, ale przy dużo mniejszej próbie lepsza może być prostsza stratyfikacja albo minimalizacja.
  \end{ExampleBox}

  \vspace{0.25cm}

  \begin{itemize}
    \item sama liczebność nie rośnie automatycznie przez liczbę warstw,
    \item rośnie jednak ryzyko, że część warstw będzie zbyt mała operacyjnie,
    \item dlatego liczebność i plan randomizacji trzeba oceniać razem, a nie osobno.
  \end{itemize}
\end{frame}

\begin{frame}{Emergency unblinding}
  \SlideID{sec05-unblinding}
  \begin{itemize}
    \item to kontrolowane ujawnienie przydziału leczenia w sytuacji medycznie uzasadnionej,
    \item nie jest zwykłą operacją administracyjną,
    \item każda taka akcja musi być uzasadniona, zapisana i możliwa do audytu.
  \end{itemize}
\end{frame}
